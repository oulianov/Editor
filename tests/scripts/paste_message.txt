# Pasting on a line of a message of several lines: the message pasted goes
# before the whole message, never between its lines.
open-event 1 1
focus commands
key End
key Return
type Show Message
key Return
focus message
type one
key Return
type two
key Return
type three
click OK
select-command 4
key Ctrl+C
select-command 4
key Ctrl+V
select-command 3
expect commands > Message: one
select-command 6
expect commands > Message: one
select-command 7
expect-not commands > Message
close
quit
