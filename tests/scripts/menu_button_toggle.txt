# A button with a drop-down menu ("... ⌄", Ez.FieldButton) works as a
# drop-down list: a click shows its menu, a click on it while the menu is
# shown closes it (not closed by the press, then shown again by the click).
# Escape, a click elsewhere close it; Down and Space show it. The clicks that
# close are made as a person's (press, wait, release): a menu closed by the
# press has the time to be closed before the release.
action actionSettings
wait 300
expect SettingsDialog Settings
expect-menu none
click linePlayerPathBrowse
wait 200
expect-menu shown
expect-enabled choosePlayer on
press linePlayerPathBrowse
wait 200
release linePlayerPathBrowse
wait 200
expect-menu none
click linePlayerPathBrowse
wait 200
expect-menu shown
# Escape closes the menu, not the window
key Escape
wait 200
expect-menu none
expect SettingsDialog Settings
# a click elsewhere
click linePlayerPathBrowse
wait 200
expect-menu shown
press linePlayerPath
wait 200
release linePlayerPath
wait 200
expect-menu none
# from the keyboard
focus linePlayerPathBrowse
key Down
wait 200
expect-menu shown
key Escape
wait 200
expect-menu none
focus linePlayerPathBrowse
key Space
wait 200
expect-menu shown
key Escape
wait 200
expect-menu none

expect-menu none
