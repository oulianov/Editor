# Edit > Search (QML window): a suggestion chosen while typing gives its
# type (a common event: the common events too); a result opens its event
# at its page and line, as in RPG Maker; the search and its results stay
# for the next time. Nothing is saved.
action actionSearch
wait 400
expect SearchDialog Search
expect label_status Enter a search query to find references in events.
expect-enabled button_search off
check-layout
focus search_text
type 1
wait 100
key Down
key Down
key Return
expect search_text 1: Switch
expect filter_type Switches
expect filter_scope Whole project
key Return
wait 300
expect label_status 11 result(s)
expect list_result 15: EV0015 | 1 | Switch | Control Switches
check-layout
# the first result, opened from the keyboard
focus list_result
key Down
key Return
wait 500
expect lineName EV0015
close
wait 200

expect label_status 11 result(s)

focus search_text
type Map Events
wait 100
key Down
key Return
expect search_text 1: Common event — Map Events
expect filter_type Event names
expect filter_scope Common events
key Return
wait 300

close
