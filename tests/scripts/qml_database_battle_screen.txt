# The Battle Screen page of the QML database window (on a copy of the
# project): the battle layout, the battle commands, a hero's own commands
# and their battle position dragged on the screen, the death handler; OK
# writes them, Cancel writes nothing.
open-project-copy
action actionDatabase
wait 300
tab tabOld_Pages 10
wait 300
expect-tab tabOld_Pages Battle screen
expect listHeroes > 0001: Zack
expect comboTroopShown 0001: Plains 1
expect-checked radioTraditional on
expect-checked radioPlacementManual on
expect-checked radioRowFront on
expect comboHeroCommand1 0001: Fight
expect comboHeroCommand3 0007: Blessing
expect listCommands > 0001: Fight
expect lineCommandName Fight
expect-checked radioDeathEvent on
expect-enabled checkDeathTeleport on
expect-enabled spinDeathX off
check-layout
resize 1280 900
check-layout

set-control radioGauge on
set-control radioWindowSmall on
set-control checkTransparent on
set-control comboHeroCommand2 3
expect comboHeroCommand2 0003: Defend
# the hero is in the front row: half a battler left of their position (264, 96)
view-drag graphicsLayout 240 96 200 120
click-item listCommands 0002: Skill
set-control lineCommandName Magic
expect listCommands > 0002: Magic
set-control comboCommandType 2
click pushAddCommand
expect listCommands > 0012: 
set-control radioDeathGameOver on
expect-enabled checkDeathTeleport off
click OK
wait 300
expect-database-file battlecommands/battle_type 2
expect-database-file battlecommands/window_size 1
expect-database-file battlecommands/transparency 1
expect-database-file battlecommands/death_handler false
expect-database-file battlecommands/death_handler_unused false
expect-database-file battlecommands/commands/1/name Magic
expect-database-file battlecommands/commands/1/type 2
expect-database-file battlecommands/commands/11/ID 12
expect-database-file actors/0/battle_commands/1 3
expect-database-file actors/0/battle_x 224
expect-database-file actors/0/battle_y 120

# Cancel: nothing written
action actionDatabase
wait 300
expect-tab tabOld_Pages Battle screen
expect-checked radioGauge on
set-control radioTraditional on
set-control radioDeathEvent on
click-item listHeroes 0002: Albert
set-control comboHeroCommand1 0
close
wait 300
expect-database-file battlecommands/battle_type 2
expect-database-file battlecommands/death_handler false
quit
