# Battle pages: the trigger of a page as one line and the window defining
# it (Define Battle Event Triggers), whose check boxes enable the fields of
# their condition. In the QML editor of the pages (the first troop of
# TestGame-2003), then from the troop page of the database. Nothing is saved.
qml-open EventControlsWindow
expect lineTrigger ""
expect commands <>
click pushTrigger
expect BattlePageConditionDialog Define Battle Event Triggers
expect-enabled comboSwitchA off
set-control checkSwitchA on
expect-enabled comboSwitchA on
expect-enabled spinTurnA off
set-control checkTurn on
expect-enabled spinTurnA on
set-control spinTurnA 3
set-control spinTurnB 1
expect-enabled spinFatigueMin off
expect-enabled comboEnemyHp off
set-control checkEnemyHp on
expect-enabled comboEnemyHp on
expect comboEnemyHp 1:Slime
set-control checkEnemyHp off
expect-enabled comboEnemyHp off
check-layout
click OK
expect lineTrigger Switch [0001:] ON
expect lineTrigger Turns [3x+1]
expect-not lineTrigger Monster
# Cancel keeps the trigger
click pushTrigger
set-control checkSwitchA off
close
expect lineTrigger Switch [0001:] ON
# the commands of the battle: inserted as in the event window
focus commands
key Return
expect EventCommandPalette Battle Event Commands
type Wait
key Return
click OK
expect commands Wait: 0.0s
close

# The troop page of the database opens the same window (on the troop
# changed in memory above)
action actionDatabase
tab tabOld_Pages 5
click pushTrigger
expect BattlePageConditionDialog Define Battle Event Triggers
set-control checkFatigue on
set-control spinFatigueMax 50
click OK
expect lineTrigger Turns [3x+1], Exhaustion [0-50]%
close
quit
