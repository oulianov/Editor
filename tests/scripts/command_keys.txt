# The keys of the command list, as in RPG Maker 2003: the arrows go from
# command to command (a message and its next lines are one), Left and Right
# as Up and Down, PgUp / PgDn by half a page; Ctrl+Tab switches pages.
# Event 1 of map 1 of TestGame-2003: a message of 2 lines, Change Level, <>.
open-event 1 1
select-command 3
key Up
expect commands > Change Level
key Up
expect commands > Message: Increase main hero 98 levels
key Up
expect commands > Message: Increase main hero 98 levels
key Down
expect commands > Change Level
key Left
expect commands > Message: Increase main hero 98 levels
key Right
expect commands > Change Level
key Right
expect commands > <>
key Right
expect commands > <>
# from the next line of a message: its first line
select-command 1
key Up
expect commands > Message: Increase main hero 98 levels
key PgDown
expect commands > <>
key PgUp
expect commands > Message: Increase main hero 98 levels
# PgDn away from the list only goes back to it
focus lineName
key PgDown
expect commands > Message: Increase main hero 98 levels
key PgDown
expect commands > <>
close
quit
