# The Terrain page of the QML database, on a copy of the project: its
# fields (a text, numbers, choices, options, a frame
# that greys what it does not use), written to the database file by OK;
# Cancel writes nothing.
open-project-copy
action actionDatabase
wait 300
tab tabOld_Pages 11
wait 300
expect-tab tabOld_Pages Terrain
expect list > 0001: Plains
expect lineName Plains
expect spinDamage 0
expect spinEncounterRate 100%
expect-checked radioBushNormal on
expect-checked radioGridDeep on
expect-checked checkAirshipPass on
expect lineFootstep (OFF)
expect lineBackdrop Grassland
check-layout

# the chance of a special battle is greyed while it is off
expect-enabled spinInitiative off
set-control checkInitiative on
expect-enabled spinInitiative on
set-control spinInitiative 150
expect spinInitiative 100%
set-control spinInitiative 40

# the frames instead of the backdrop (RPG Maker 2003)
expect-enabled groupFrameBack off
expect-enabled lineBackdrop on
set-control radioFrame on
expect-checked radioBackground off
expect-enabled groupFrameBack on
expect-enabled lineBackdrop off
expect-enabled lineFrameFore off
set-control checkFrameFore on
expect-enabled lineFrameFore on
expect-enabled spinFrameForeScrollH off
set-control checkFrameForeScrollH on
set-control spinFrameForeScrollH -12
expect spinFrameForeScrollH -8

set-control lineName Meadow
expect list > 0001: Meadow
set-control spinDamage -1500
expect spinDamage -999
set-control spinEncounterRate 250
set-control radioBushHalf on
set-control radioGridShallow on
set-control checkBoatPass on
set-control checkAirshipLand off
click OK
wait 300
expect-database-file terrains/0/name Meadow
expect-database-file terrains/0/damage -999
expect-database-file terrains/0/encounter_rate 250
expect-database-file terrains/0/bush_depth 2
expect-database-file terrains/0/grid_location 0
expect-database-file terrains/0/boat_pass true
expect-database-file terrains/0/airship_land false
expect-database-file terrains/0/special_flags/back_party true
expect-database-file terrains/0/special_back_party 40
expect-database-file terrains/0/background_type 1
expect-database-file terrains/0/background_b true
expect-database-file terrains/0/background_b_scrollh true
expect-database-file terrains/0/background_b_scrollh_speed -8

# Cancel: nothing written
action actionDatabase
wait 300
expect-tab tabOld_Pages Terrain
expect lineName Meadow
expect-checked radioFrame on
set-control lineName Nothing
set-control spinDamage 5
set-control checkShipPass on
close
wait 300
expect-database-file terrains/0/name Meadow
expect-database-file terrains/0/damage -999
expect-database-file terrains/0/ship_pass false
quit
