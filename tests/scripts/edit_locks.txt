# A modal window holds an edit lock while it is open (ModalWindowLocks,
# QmlHost::exec): what another program (rpgsync) writes into the project is
# read again only once it is closed, never under a dialog that can keep
# pointers into the data. The event and database windows hold their own.
# On a copy of the project.
open-project-copy
expect-locks none

# the Settings: changed files are read again (in the script's own settings)
action actionSettings
wait 300
expect-locks dialog
set-control checkReloadExternal on
click OK
wait 200
expect-locks none

# the database changed on disk while Map Properties is open: read once it
# is closed
select-map 1
action actionMapProperties
wait 300
expect-locks dialog
edit-database-file terms/victory Locked out
wait 1500
expect-database terms/victory Victory!
close
wait 300
expect-locks none
wait 1500
expect-database terms/victory Locked out
# without a dialog: read at once
edit-database-file terms/victory Read at once
wait 1500
expect-database terms/victory Read at once

# a question of the controllers (DialogService)
dialog-service question "Title" "A question"
wait 200
expect-locks dialog
click No
wait 200
expect-answer no
expect-locks none

# a picker
pick-entry switches 1
wait 300
expect-locks dialog
close
wait 200
expect-answer cancel
expect-locks none

# the event window: its own lock, the maps are still read again; a
# command chosen in it holds a dialog's
open-event 1 1
wait 300
expect-locks event:1
focus commands
key End
key Return
wait 300
expect-locks event:1 dialog
close
wait 200
expect-locks event:1
close
wait 200
expect-locks none

# the database window: its own lock
action actionDatabase
wait 300
expect-locks database
close
wait 200
expect-locks none
quit
