# The Hero Status page of the QML database, on a copy of the project: its
# fields (most of Death fixed, the messages of RPG
# Maker 2000 greyed in a 2003 game), written to the database file by OK;
# Cancel writes nothing.
open-project-copy
action actionDatabase
wait 300
tab tabOld_Pages 7
wait 300
expect-tab tabOld_Pages Hero Status
expect list > 0001: Death
expect lineName Death
expect comboExtents Persists after battle
expect comboBattlerPose 05:
expect-enabled lineName on
expect-enabled spinRateA on
expect-enabled spinPriority off
expect-enabled comboColor off
expect-enabled radioHPEffectNone off
expect-enabled lineMessageInflictionActor off
check-layout

# another condition: its fields can be changed
click-item list 0002: Poison
expect lineName Poison
expect-enabled spinPriority on
expect-enabled radioHPEffectNone on
expect-enabled comboBattlerPose on
expect-enabled lineMessagePerTurn off
set-control lineName Venom
expect list > 0002: Venom
set-control comboExtents 1
set-control comboColor 3
expect comboColor Color 3
set-control spinPriority 150
expect spinPriority 100
set-control comboRestriction 2
set-control spinAccuracy 75
set-control checkReflectSkills on
set-control spinRateB 55
set-control checkRestrictAttackInfluence on
set-control spinRestrictAttackInfluence 4
set-control radioStatAlterationDouble on
expect-checked radioStatAlterationHalf off
set-control checkStatAffectAgility on
set-control spinRecoverTurns 3
set-control spinRecoverChance 25
set-control radioHPEffectRecovery on
set-control spinHPEffectValue 12
set-control spinHPEffectFieldSteps 8
set-control radioMPEffectNone on
set-control comboBattlerPose 6
expect comboBattlerPose 07:
click OK
wait 300
expect-database-file states/1/name Venom
expect-database-file states/1/type 1
expect-database-file states/1/color 3
expect-database-file states/1/priority 100
expect-database-file states/1/restriction 2
expect-database-file states/1/reduce_hit_ratio 75
expect-database-file states/1/reflect_magic true
expect-database-file states/1/b_rate 55
expect-database-file states/1/restrict_skill true
expect-database-file states/1/restrict_skill_level 4
expect-database-file states/1/affect_type 1
expect-database-file states/1/affect_agility true
expect-database-file states/1/hold_turn 3
expect-database-file states/1/auto_release_prob 25
expect-database-file states/1/hp_change_type 1
expect-database-file states/1/hp_change_val 12
expect-database-file states/1/hp_change_map_steps 8
expect-database-file states/1/sp_change_type 2
expect-database-file states/1/battler_animation_id 6

# Cancel: nothing written
action actionDatabase
wait 300
expect-tab tabOld_Pages Hero Status
expect lineName Venom
set-control lineName Nothing
set-control spinRateB 5
set-control radioHPEffectDamage on
close
wait 300
expect-database-file states/1/name Venom
expect-database-file states/1/b_rate 55
expect-database-file states/1/hp_change_type 1

# RPG Maker 2000: the battle messages, no pose nor RPG Maker 2003 effects
open-project-copy
edit-database-file system/ldb_id 0
open-project-copy
action actionDatabase
wait 300
tab tabOld_Pages 7
wait 300
click-item list 0002: Venom
expect-enabled lineMessageInflictionActor on
expect-enabled lineMessagePerTurn on
expect-enabled comboBattlerPose off
expect-enabled checkEvadeAll off
expect-enabled radioHPEffectNone off
close
quit
