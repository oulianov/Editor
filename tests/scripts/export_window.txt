# Game Project > Export Game (QML window), on a copy of the game in the
# script's own folder (the export settings are written in it): Windows,
# with the game's RPG_RT.exe as its Player (nothing downloaded), into a
# folder of the script, with its log and its progress.
open-copy
wait 300
action actionGameDiskCreate
wait 500
expect ExportDialog Export Game
expect gameTitle RPG Maker 2003 Test suite
expect exportProgress 0/1
check-layout
set-control export_macos off
set-control export_linux off
set-control export_web off
set-control export_windows on
expect-enabled path_windows on
expect-enabled path_macos off
expect-enabled customizeApp off
click launcher_windows
wait 200
menu RPG_RT.exe of the game
wait 200
expect launcher_windows RPG_RT.exe of the game
set-control path_windows $SCRIPT_DIR/export/windows
set-control makeZip off
expect-enabled openFolder off
expect-enabled cancelExport off
click exportButton
wait 8000
expect exportLog == Windows
expect exportLog Export finished.
expect-not exportProgress 0/1
expect-enabled openFolder on
expect-enabled exportButton on
expect-enabled cancelExport off
check-layout
close
quit
