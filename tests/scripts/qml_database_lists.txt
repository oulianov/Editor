# The lists of the QML database: the switches and variables alone (as the
# windows of the "..." buttons), the search over the numbers and names, the
# menu of a list (Copy, Paste, Clear, Cut, Multiple Copy), Array Size and
# F4/F5. On a copy of the project; what OK writes is checked in the file.
open-project-copy

# The switches: the name of an entry
qml-open DatabaseWindow list=switches
wait 300
expect DatabaseDialog Switches
expect list > 0001: 
click-item list 0010: ShowRoomDesc
expect lineName ShowRoomDesc
set-control lineName Room shown
expect list > 0010: Room shown
check-layout

# the search: the entry of a number typed (with or without its zeros), then
# every word anywhere in the name; the entry shown stays while it matches
focus listFilter
type 12
expect list > 0012: TestBattleOn
expect lineName TestBattleOn
set-control listFilter ""
focus listFilter
type 0016
expect list > 0016: AntilagAutorun
set-control listFilter ""
focus listFilter
type antilag a
expect list > 0016: AntilagAutorun
expect list 0015: AntilagParallel
expect-not list 0012
focus list
key Up
expect list > 0015: AntilagParallel
set-control listFilter zzz
expect lineName AntilagParallel
set-control listFilter ""
expect list > 0015: AntilagParallel
click OK
wait 300
expect-database-file switches/9/name Room shown

# The variables
qml-open DatabaseWindow list=variables
wait 300
expect DatabaseDialog Variables
focus listFilter
type 13
expect list > 0013: Player Map
set-control lineName Counter
close
wait 300
expect-database-file variables/12/name Player Map
qml-open DatabaseWindow list=variables
wait 300
expect list > 0013: Player Map
set-control lineName Counter
click Apply
wait 300
expect-database-file variables/12/name Counter
close
wait 300

# The menu of a list, in the database window: Copy and Paste keep the IDs
action actionDatabase

menu Copy
wait 300

wait 300

focus lineName
key F5

key F4
key F4
expect list > 0004: Bow
# Clear (Delete in the list), Cut
focus list
key Del
expect list > 0004: 
expect lineName ""
click-item list 0002: Spear
action actionDataCut
expect list > 0002: 
click-item list 0004: 
action actionDataPaste
expect list > 0004: Spear
# Multiple Copy: from the entry shown on
click-item list 0007: Thunder
context-menu list

menu Multiple Copy

wait 300
click-item list 0010: Wind

# Array Size: entries added at the end; fewer warns about those with data
click pushMax

wait 300

wait 300
expect messageText Reducing the array size from 20 to 3 deletes 17 entries, 14 of them with data:
expect messageText 0017: Absorb
check-layout
click Cancel
wait 300
expect list 0020: 
click pushMax
wait 300
set-control spinArraySize 18
click OK
wait 300
expect-not list 0019: 
expect list 0018: 
# nothing of it reaches the file: Cancel
close

expect-database-file attributes/4/name Fire
