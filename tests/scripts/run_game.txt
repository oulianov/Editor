# Game > Test Play Options (QML window), on a copy of the game in the
# script's own folder (a battle test saves its party in the database): the
# play modes enable their groups, a battle test shows the party of the
# battle tests (TestBattlePartyView). There is no EasyRPG Player here: OK
# offers one (download or locate), and the options stay open when none
# starts. The arguments the Player is given: test_play_options_test.
open-copy
wait 300
expect-player none
action actionPlayTestOptions
wait 500
expect RunGameDialog Test Play Options
expect comboMode Normal
expect comboDisplayMode Window
expect comboTroop Plains 1
expect comboMapId 0001: Map events
expect comboInitialParty1 Zack
expect-enabled groupInitialPosition on
expect-enabled groupInitialParty on
expect-enabled comboMapId off
expect-enabled groupSave off
expect-enabled groupBattleOptions off
expect-checked radioCustomFormation on
check-layout
set-control groupInitialPosition on
expect-enabled comboMapId on
set-control spinX 12
expect spinX X: 12

# a savegame: no start position nor party
set-control comboMode 2
wait 100
expect-enabled groupSave on
expect-enabled groupInitialPosition off
expect-enabled groupInitialParty off
expect-enabled groupBattleOptions off

# a battle test: its options and party
set-control comboMode 3
wait 200
expect-enabled groupSave off
expect-enabled groupBattleOptions on
expect-enabled comboTroop on
check-layout
set-control radioPartySize2 on
expect-enabled comboHero1 on
expect-enabled comboHero2 on
expect-enabled comboHero3 off
expect comboHero1 0001: Zack
set-control spinLevel1 10
expect spinLevel1 10
expect comboWeapon1 Club
set-control comboWeapon1 0
expect comboWeapon1 <None>
click pushInitialStats1
expect spinLevel1 1
expect comboWeapon1 Club

# OK: the party saved, then no Player to start
click OK
wait 500
expect playerMissing Test Play
close
wait 300
expect RunGameDialog Test Play Options
close
wait 300

# the party of the database, as saved
action actionPlayTestOptions
wait 500
set-control comboMode 3
wait 200
expect-checked radioPartySize2 on
expect-enabled comboHero3 off
close
quit
