# The questions and messages of the QML DialogService (MessageWindow): each
# kind shows its title, its text and its buttons, and gives the answer of the
# button clicked by its text (Escape and the close button: no). Nothing is
# changed.

# a file changed on disk: reload, save mine, cancel
dialog-service stale "The map MAP0001.lmu" "Hero 1" "Hero 2"
expect-answer (waiting)
expect messageWindow Changed on Disk
expect messageText The map MAP0001.lmu was changed by another program since the Editor read it.
expect messageText Changed on both sides: Hero 1, Hero 2.
expect buttonBox Reload (Discard Mine)
expect buttonBox Save Mine
expect buttonBox Cancel
check-layout
click Reload (Discard Mine)
expect-answer reload
dialog-service stale "The database"
expect-not messageText Changed on both sides
click Save Mine
expect-answer overwrite
dialog-service stale "The database"
click Cancel
expect-answer cancel
dialog-service stale "The database"
key Escape
expect-answer cancel
# what cannot be read again (an event being edited): no reload
dialog-service stale-no-reload "This event"
expect-not buttonBox Reload
expect buttonBox Save Mine
click Save Mine
expect-answer overwrite

# questions: Yes and No, and Cancel when asked
dialog-service question "Delete Map" "Delete the map and its events?"
expect messageWindow Delete Map
expect messageText Delete the map and its events?
expect-not buttonBox Cancel
check-layout
click Yes
expect-answer yes
dialog-service question "Delete Map" "Delete the map and its events?"
click No
expect-answer no
dialog-service question "Delete Map" "Delete the map and its events?"
close
expect-answer no
dialog-service question-cancel "Unsaved Changes" "Save the changes of the map?"
expect buttonBox Cancel
click No
expect-answer no
dialog-service question-cancel "Unsaved Changes" "Save the changes of the map?"
click Cancel
expect-answer cancel
dialog-service question-cancel "Unsaved Changes" "Save the changes of the map?"
click Yes
expect-answer yes

# messages: OK
dialog-service information "Export" "The game was exported."
expect messageWindow Export
expect messageText The game was exported.
expect buttonBox OK
click OK
expect-answer ok
dialog-service warning "Import" "Some files could not be read."
expect messageText Some files could not be read.
key Escape
expect-answer ok
dialog-service error "Save" "The map could not be saved: the disk is full. The changes stay in the Editor until they can be saved."
check-layout
close
expect-answer ok
quit
