# The palette of the QML map window (Palette.qml): its tiles picked with
# the mouse (a rectangle at most 6 rows
# high, the eraser row alone, the right button cancels), its frame on the
# tiles picked, also by the eyedropper of the map, and each layer's.
# Run on TestGame-2003, map 1; nothing is saved.
qml-map 1
action actionLayerLower
action actionDrawPen
# a tile, then a rectangle of them
palette-click 0 1
expect-palette 0 1 1 1
expect-picked 47 1 1
palette-drag 1 2 3 3
expect-palette 1 2 3 2
expect-picked 4097 3 2
map-move 5 5
expect-brush 5 5 3 2
# at most 6 rows high, from the cell the drag started on
palette-drag 0 4 0 12
expect-palette 0 4 1 6
palette-drag 0 12 0 4
expect-palette 0 7 1 6
# the eraser row is picked alone: water on the lower layer
palette-drag 2 0 4 3
expect-palette 0 0 6 1
expect-picked 47 1 1
# a drag from the tiles stops before the eraser row
palette-drag 2 3 2 0
expect-palette 2 1 1 3
# the right button cancels a drag: the tiles picked stay
palette-drag 1 5 3 6 cancel
expect-palette 2 1 1 3
expect-picked 2047 1 3
# the eyedropper of the map (a right click) frames the tile it picks
map-click 6 6 right
expect-palette 0 2 1 1
# the upper layer: its tiles and its own tiles picked; its eraser, the
# transparent tile
action actionLayerUpper
expect-palette 0 1 1 1
palette-click 4 2
expect-picked 10010 1 1
palette-click 3 0
expect-picked 10000 1 1
expect-palette 0 0 6 1
# the event layer shows them as they are
action actionLayerEvents
expect-palette 0 0 6 1
action actionLayerLower
expect-palette 0 2 1 1
# Ctrl + wheel zooms the palette, by whole pixels of its tiles: its cells
# are picked as before
expect-palette-zoom 2
palette-wheel 1 1 120 ctrl
expect-palette-zoom 4
palette-wheel 1 1 120 ctrl
expect-palette-zoom 4
palette-click 2 3
expect-palette 2 3 1 1
expect-picked 4447 1 1
palette-wheel 1 1 -240 ctrl
expect-palette-zoom 1
quit
