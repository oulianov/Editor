# The pickers of the QML interface (Pickers::pickFile and pickEntry: the
# FilePicker and EntryPicker windows) and the fields opening them
# (Ez.SpriteField, Ez.AudioField, Ez.FileField of PickerFieldsWindow, on the
# first hero, monster and terrain and the system of TestGame-2003): the file
# or entry selected first, what OK, Apply and Cancel give. Nothing is saved.

# without an RTP, as in CI: its files are not listed
action actionSettings
wait 300
set-control lineRtpPath ""
click OK
wait 200

# A file is always selected: the current one (any case)...
pick-file CharSet CharSet actor4 2
wait 300
expect PickerDialog Select from CharSet
expect filesystemView > Actor4.png
expect labelInfo Game file: Actor4.png
check-layout
click OK
wait 200
expect-answer Actor4:2:2:1:0:0:100:100:50
# ...else the first of the list: no file yet, or a missing one
pick-file CharSet CharSetPose ""
wait 300
expect filesystemView > Actor1.png
# Cancel: no choice
close
wait 200
expect-answer cancel
pick-file CharSet CharSetPose Missing
wait 300
expect filesystemView > Actor1.png
expect-checked radioFacing2 on
expect-checked radioFrame1 on
check-layout
click radioFacing3
click radioFrame0
click Actor3.png
click OK
wait 200
expect-answer Actor3:0:3:0:0:0:100:100:50
# Apply gives the file while the picker stays; Cancel then keeps it given
pick-file Panorama Image ""
wait 300
expect filesystemView > planet1.png
click Apply
wait 100
expect PickerDialog Select from Panorama
close
wait 200
expect-answer applied:planet1:0:2:1:0:0:100:100:50
# a sound, heard with its settings: they come back with it
pick-file Sound Sound Cat
wait 300
expect filesystemView > Cat.wav
expect labelVolume 100%
expect-enabled resetVolume off
set-control sliderVolume 40
set-control sliderBalance 30
expect labelVolume 40%
expect labelBalance 70:30
expect-enabled buttonStop off
check-layout
click OK
wait 200
expect-answer Cat:0:2:1:0:0:40:100:30
# RPG Maker's steps (its trackbars: 11 positions): a value of the data
# between two steps stays as it is until the slider is moved
pick-file Sound Sound Cat volume=81 tempo=73 balance=47
wait 300
expect labelVolume 81%
expect labelTempo 73%
expect labelBalance 53:47
expect-enabled resetVolume on
click OK
wait 200
expect-answer Cat:0:2:1:0:0:81:73:47
pick-file Sound Sound Cat volume=81 tempo=73 balance=47
wait 300
# the wheel only once the slider has the focus (the window scrolls)
view-wheel sliderVolume 50 8 120
expect labelVolume 81%
# the arrow keys: to the next step, Home and End: to the ends
focus sliderVolume
key Right
expect labelVolume 90%
key Left
expect labelVolume 80%
key Left
expect labelVolume 70%
key End
expect labelVolume 100%
# the default value: its reset button greyed
expect-enabled resetVolume off
key Home
expect labelVolume 0%
# a notch of the wheel: a step (half notches of a touchpad add up)
view-wheel sliderVolume 50 8 120
expect labelVolume 10%
view-wheel sliderVolume 50 8 60
expect labelVolume 10%
view-wheel sliderVolume 50 8 60
expect labelVolume 20%
# dragged: on a step (42% of the way: 90%, not 92%)
view-drag sliderTempo 60 8 103 8
expect labelTempo 90%
# reset: the default value of RPG Maker (lcf::rpg::Sound), the button greyed
click resetVolume
expect labelVolume 100%
expect-enabled resetVolume off
click resetTempo
expect labelTempo 100%
expect-enabled resetTempo off
expect-enabled resetBalance on
click resetBalance
expect labelBalance 50:50
expect-enabled resetBalance off
click OK
wait 200
expect-answer Cat:0:2:1:0:0:100:100:50
# a music: its fade-in too, in seconds (2.5 s kept, reset to none)
pick-file Music Music "" fadein=2500 volume=40
wait 300
expect labelFadeIn 2.5s
expect labelVolume 40%
check-layout
click resetFadeIn
expect labelFadeIn 0s
expect-enabled resetFadeIn off
click OK
wait 200
expect-answer "Battle 1:0:2:1:0:0:40:100:50"
# an empty folder (no monster in the game): nothing to give
pick-file Monster Monster Slime
wait 300
expect-enabled OK off
expect-enabled Apply off
close
wait 200
expect-answer cancel

# Entries: the current one, "(None)" first where the field offers it
pick-entry switches 3
wait 300
expect EntryPicker Switches
expect list > 0003
expect-not list (None)
check-layout
close
wait 200
expect-answer cancel
pick-entry switches 0 none
wait 300
expect list > (None)
click OK
wait 200
expect-answer 0
# searched by number or name: the entry found is selected
pick-entry variables 2 none
wait 300
expect list > 0002
set-control listFilter 12
expect list > 0012
# the entry selected stays while it matches
set-control listFilter 1
expect list > 0012
click OK
wait 200
expect-answer 12
# a name changed in the picker: in the project on OK, and on Apply
pick-entry switches 1
wait 300
set-control lineName Door open

wait 200
expect-answer 1
pick-entry switches 2
wait 300
expect list 0001: Door open
set-control lineName Applied

close

wait 200
