# The map tabs (MapTabsController): a map only looked at opens in the
# temporary tab (~, in italics), which the next one replaces; a double-click
# or an edit keeps it. Closing a tab with changes asks first: Cancel keeps
# it, No drops them. Nothing is saved.
# Run on TestGame-2003 (https://github.com/EasyRPG/TestGame): it opens on
# its active map, "call".
expect tabMap call
open-map 2
expect tabMap call | ~Picture events
open-map 1
expect tabMap call | ~Map events
# an edit keeps it; the next map opens in a new temporary tab
action actionLayerLower
action actionDrawPen
palette 7
map-click 1 1
expect tabMap call | Map events *
open-map 2
expect tabMap call | Map events * | ~Picture events
double-click-item treeMap "Picture events"
expect tabMap call | Map events * | Picture events
open-map 4
expect tabMap call | Map events * | Picture events | ~Generate Dungeon

# a tab dragged elsewhere stays shown
move-tab tabMap 3 0
expect tabMap ~Generate Dungeon | call | Map events * | Picture events
expect-tab tabMap "Generate Dungeon"

# closing: Cancel keeps the changes, No drops them
close-tab tabMap 2
click Cancel
wait 200
expect tabMap ~Generate Dungeon | call | Map events * | Picture events
close-tab tabMap 2
click No
wait 200
expect tabMap ~Generate Dungeon | call | Picture events
expect-enabled actionMapSave off
# the tab shown closed: the one on its right is shown
close-tab tabMap 0
wait 200
expect tabMap call | Picture events
expect-tab tabMap "call"
quit
