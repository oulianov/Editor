# The conditions of an event page and the rules of RPG Maker 2003's event
# window: a condition's fields are enabled while it is checked, the
# comparison of the variable and the second timer are RPG Maker 2003's,
# the frequency is not for an event standing still, the pattern is the
# Custom Pattern's, and a page without a character does not animate.
# Event 1 of map 1 of TestGame-2003 (a RPG Maker 2003 game). Nothing is saved.
open-event 1 1
expect-enabled comboSwitchA off
set-control checkSwitchA on
expect-enabled comboSwitchA on
set-control checkSwitchA off
expect-enabled comboSwitchA off
expect-enabled comboSwitchB off
set-control checkSwitchB on
expect-enabled comboSwitchB on

expect-enabled comboVariable off
expect-enabled comboVarOperation off
expect-enabled spinVarValue off
set-control checkVar on
expect-enabled comboVariable on
expect-enabled comboVarOperation on
expect-enabled spinVarValue on
expect comboVarOperation Greater or equal to (>=)
set-control comboVarOperation 0
expect comboVarOperation Equal to (==)
set-control spinVarValue 12
expect spinVarValue 12

expect-enabled comboItem off
set-control checkItem on
expect-enabled comboItem on
expect-enabled comboHero off
set-control checkHero on
expect-enabled comboHero on

expect-enabled spinTimerAMin off
set-control checkTimerA on
expect-enabled spinTimerAMin on
expect-enabled spinTimerASec on
expect-enabled checkTimerB on
set-control checkTimerB on
expect-enabled spinTimerBSec on

# the movement and the graphic
expect comboMoveType Stationary
expect-enabled comboMoveFrequency off
set-control comboMoveType 1
expect comboMoveType Random
expect-enabled comboMoveFrequency on
expect-enabled pushMovePattern off
set-control comboMoveType 6
expect-enabled pushMovePattern on
expect comboCondition Action Key
expect comboAnimationType Non-Continuous
expect-enabled comboAnimationType on
click pushSetSprite
menu No Graphic
expect-enabled comboAnimationType off
check-layout

# what was set is kept, as applied
click Apply
close
open-event 1 1
expect-checked checkSwitchB on
expect-checked checkVar on
expect comboVarOperation Equal to (==)
expect spinVarValue 12
expect-checked checkTimerB on
expect comboMoveType Custom Pattern
expect-enabled comboAnimationType off
close
quit
