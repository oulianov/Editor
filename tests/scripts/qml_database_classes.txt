# The Professions page of the QML database, on a copy of the project: its
# fields (a name, options, battle graphics and
# commands, resistances, the skills learned, the experience curve in its
# window), written to the database file by OK; Cancel writes nothing.
open-project-copy
action actionDatabase
wait 300
tab tabOld_Pages 1
wait 300
expect-tab tabOld_Pages Professions
expect list > 0001: Hero A
expect lineName Hero A
expect comboBattler <none>
expect comboBattleCommand3 0007: Blessing
expect tableSkills 4 | Heal Force 1
expect listStates rank2 | Death
expect labelExpCurve Primary: 1, Secondary: 677, Tertiary: 40
check-layout

set-control lineName Knight
expect list > 0001: Knight
set-control checkTwoWeapons on
set-control checkMightyGuard on
set-control comboBattler 2
expect comboBattler 0002:
# a battle command removed: the others move up, the Row command after them
set-control comboBattleCommand2 0
# a click (Space) gives a resistance the next rank
focus listStates
key Down
key Space
key Space
expect listStates > rank4 | Death
focus listAttributes
key Down
key Down
key Space
expect listAttributes > rank3 | Spear
# Delete removes a skill learned
focus tableSkills
key Down
key Del
expect-not tableSkills Teleport

# the experience curve in its window: OK writes its settings
click pushExpCurveEdit
wait 400
expect ExperienceDialog Experience Curve
expect spinExpBase 1
expect tableTotalExperience 2 |
set-control spinExpBase 30
set-control spinExpCorrection 10
click OK
wait 300
expect labelExpCurve Primary: 30, Secondary: 677, Tertiary: 10
# Cancel leaves it
click pushExpCurveEdit
wait 400
set-control spinExpInflation 5
click Cancel
wait 300
expect labelExpCurve Primary: 30, Secondary: 677, Tertiary: 10

click OK
wait 300
expect-database-file classes/0/name Knight
expect-database-file classes/0/two_weapon true
expect-database-file classes/0/super_guard true
expect-database-file classes/0/battler_animation 2
expect-database-file classes/0/battle_commands/0 1
expect-database-file classes/0/battle_commands/1 7
expect-database-file classes/0/battle_commands/4 5
expect-database-file classes/0/battle_commands/5 0
expect-database-file classes/0/battle_commands/6 -1
expect-database-file classes/0/state_ranks/0 4
expect-database-file classes/0/attribute_ranks/1 3
expect-database-file classes/0/skills/0/level 2
expect-database-file classes/0/exp_base 30
expect-database-file classes/0/exp_inflation 677
expect-database-file classes/0/exp_correction 10

# Cancel: nothing written
action actionDatabase
wait 300
expect-tab tabOld_Pages Professions
expect lineName Knight
set-control lineName Nothing
set-control checkAutoBattle on
set-control comboBattleCommand1 0
close
wait 300
expect-database-file classes/0/name Knight
expect-database-file classes/0/auto_battle false
expect-database-file classes/0/battle_commands/0 1
quit
