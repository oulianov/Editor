# The Chipset page of the QML database window (on a copy of the project):
# the tiles clicked and dragged in each editing mode (passability, RPG
# Maker 2003's directions, terrain, Set Uniform Terrain) on both layers,
# the water animation; OK writes them, Cancel writes nothing.
open-project-copy
action actionDatabase
wait 300
tab tabOld_Pages 12
wait 300
expect-tab tabOld_Pages Chipset
expect list > 0001: World Map
expect lineName World Map
expect lineChipsetFile World
expect-checked radioPassability on
expect-enabled comboTerrain off
expect-tab tabLayer Lower Layer
check-layout
resize 1280 900
check-layout

set-control lineName Overworld
# passable -> blocked; a drag gives the same to the other tiles
view-click viewTiles 48 48
view-drag viewTiles 80 48 112 48
# the top side of a tile
set-control radioDirections on
view-click viewTiles 48 34
# a terrain for every lower tile, then one tile
set-control radioTerrain on
expect-enabled comboTerrain on
click pushUniformTerrain
expect SetUniformTerrainDialog Set Uniform Terrain
set-control comboUniformTerrain 1
click OK
set-control comboTerrain 2
view-click viewTiles 16 16
set-control radioSpeedFast on
# the upper layer: blocked -> above the hero
tab tabLayer 1
wait 100
expect-enabled pushUniformTerrain off
set-control radioPassability on
view-click viewTiles 48 16
click OK
wait 300
expect-database-file chipsets/0/name Overworld
expect-database-file chipsets/0/passable_data_lower/7 8
expect-database-file chipsets/0/passable_data_lower/8 0
expect-database-file chipsets/0/passable_data_lower/9 0
expect-database-file chipsets/0/terrain_data/0 3
expect-database-file chipsets/0/terrain_data/20 2
expect-database-file chipsets/0/animation_speed 1
expect-database-file chipsets/0/passable_data_upper/1 31

# Cancel: nothing written
action actionDatabase
wait 300
expect-tab tabOld_Pages Chipset
set-control lineName Nothing
tab tabLayer 0
set-control radioCounter on
view-click viewTiles 48 48
close
wait 300
expect-database-file chipsets/0/name Overworld
expect-database-file chipsets/0/passable_data_lower/7 8
quit
