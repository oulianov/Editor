# Opening a project while one is open closes it first: its map tabs, after
# asking about their unsaved changes (Cancel keeps it open), so no tab is
# left on the data of the project closed. Reading the map tree again
# afterwards (another program changed it) refreshes the tabs of the new one.
# Run on TestGame-2003 (https://github.com/EasyRPG/TestGame).
open-map 1
open-map 4
action actionDrawPen
action actionLayerLower
palette 7
map-click 1 1
expect-enabled actionMapSave on

# Cancel: the project stays open, with its changes
open-project
click Cancel
wait 200
expect-enabled actionMapSave on

# No: the changes are dropped, the project opened again
open-project
click No
wait 300
expect treeMap RPG Maker 2003 Test suite
expect-enabled actionMapSave off
reload-tree
wait 200
expect treeMap Dungeon Room
open-map 1
reload-tree
wait 200
expect-enabled actionMapProperties on
quit
