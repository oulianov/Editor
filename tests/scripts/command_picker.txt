# The command picker and the move route window
open-event 1 1
focus commands
key End
key Space
check-layout
type Move Event
key Return
click Move Up
# a move with a value (here a switch) between two others: the list must
# decode it like the window, not read its value as the next move
click Switch ON
wait 200
click OK
click Face Hero
check-layout
click OK
expect commands "Move Event: Hero, Move Up, Switch ON: [0001"
expect commands "], Face Hero"
# The search shows the matches of every page: "cond" finds Change Condition
# and Conditional Branch, whatever page is shown
focus commands
key End
key Return
type cond
expect-enabled Change Condition on
click Conditional Branch
wait 200
click OK
expect commands "Branch if"
close
quit
