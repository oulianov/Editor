# Tools > Resource Manager (QML) on a copy of the game: the folders, the
# files of a folder and what the one selected is, the zoom of the preview,
# Export, then Import of the file exported into a folder the game does not
# have yet (its size questioned, its transparent color chosen), Rename and
# Delete. Only the game's files are checked: scripts may run without an RTP.
open-copy
action actionResourceManager
wait 500
expect ResourceManagerDialog Resource Manager
expect buttonBox Close
expect buttonBox Help
check-layout

# the folders of RPG Maker 2003, the files of one
expect listFolders CharSet
click CharSet
expect listFolders > CharSet
click Actor1
expect listFiles > Actor1
expect previewInfo Game file: Actor1.png
expect previewInfo 288 x 256 pixels
expect-enabled pushExport on
expect-enabled pushRename on
expect-enabled pushDelete on
expect-enabled radioImportPng on
expect-checked radioImportPng on
expect-enabled pushZoomOut off
click pushZoomIn
expect-enabled pushZoomOut on
click pushZoomOut
expect-enabled pushZoomOut off
set-control filterFiles Actor2
expect-not listFiles Actor1
set-control filterFiles ""
expect listFiles Actor1

# Export: a copy of the file
call ResourceManagerDialog exportTo $SCRIPT_DIR/Hero.png
wait 200

# Import into Frame, a folder the game has not: made, the size questioned
# (not the 320 x 240 of a frame), the transparent color chosen
click Frame
expect listFolders > Frame
expect-not listFiles Hero
call ResourceManagerDialog importFiles $SCRIPT_DIR/Hero.png
wait 400
expect messageText RPG Maker's Frame images are 320 x 240 pixels; this one is 288 x 256.
click Yes
wait 400
expect ImportImageDialog Transparent color of Hero.png
expect labelHelp Click the color that is transparent in the game.
expect labelColor (palette index 0)
check-layout
# small game graphics open enlarged
expect-enabled pushZoomOut on
click pushZoomOut
expect-enabled pushZoomOut off
call ImportImageDialog pick 12 16
expect-not labelColor (palette index 0)
call ImportImageDialog pick 0 0
expect labelColor (palette index 0)
click OK
wait 400
expect ResourceManagerDialog Resource Manager
expect listFiles > Hero
expect previewInfo Game file: Hero.png
expect previewInfo Palette image

# the same name again: replaced only when asked
call ResourceManagerDialog importFiles $SCRIPT_DIR/Hero.png
wait 400
expect messageText RPG Maker's Frame images are 320 x 240 pixels
click Yes
wait 400
click Cancel
wait 300
expect listFiles > Hero

# Rename, keeping its extension
click pushRename
wait 200
expect lineNewName Hero
set-control lineNewName Heroine
click OK
wait 300
expect listFiles > Heroine
expect previewInfo Game file: Heroine.png
expect-not listFiles Hero |

# Delete, after a question
click pushDelete
wait 300
expect messageText Delete Heroine.png from the Frame folder of the game?
click No
wait 200
expect listFiles > Heroine
click pushDelete
wait 300
click Yes
wait 300
expect-not listFiles Heroine
expect-enabled pushDelete off

# sounds: no picture (not played: no sound in the tests)
click Music
click Cat
expect listFiles > Cat
expect previewImage Press Play, or double-click the file, to listen to it.
expect previewInfo Format: WAV
expect-enabled pushPlay on
close
wait 300

# the image import window alone, on a file (import-image-dialog)
import-image-dialog $SCRIPT_DIR/Hero.png
wait 400
expect ImportImageDialog Select transparent color
expect labelColor Transparent color:
check-layout
close
quit
