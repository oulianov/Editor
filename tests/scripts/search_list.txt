# The lists of database entries are searched by number or name, in their
# drop-down (F4; typing on a box with "..." renames its entry instead).
open-event 1 1
focus commands
key End
key Return
type Control Variables
key Return

# By number: the entry of the number typed, leading zeros or not
focus targetVariable
key F4
type 12
key Return
expect targetVariable 0012
focus targetVariable
key F4
type 0003
key Return
expect targetVariable 0003

# By name, each word anywhere in it; the arrows move in what matches
focus targetVariable
key F4
type player m
key Return
expect targetVariable 0013
focus targetVariable
key F4
type var
key Down
key Return
expect targetVariable 0002

# Nothing matches, or Escape: the entry stays
focus targetVariable
key F4
type zzz
key Return
key Escape
expect targetVariable 0002
focus targetVariable
key Space
type 1
key Escape
expect targetVariable 0002
close

# The events of the map (Move Event and the other commands on events)
focus commands
key End
key Return
type Move Event
key Return
focus moveTarget
type ev0012
key Return
expect moveTarget 0012: EV0012
focus moveTarget
type airship
key Return
expect moveTarget Airship
close
close
quit
