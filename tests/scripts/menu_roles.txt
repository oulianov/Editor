# macOS's application menu: Quit, About and Settings by their roles, Test
# Play Options copied beside Settings; each one in its menu of the menu
# bar too, and each one runs from the native menu bar. That menu bar is not
# drawn, nor shown offscreen: its items are checked (menu-role).
# Run on TestGame-2003 (https://github.com/EasyRPG/TestGame). Nothing is saved.
menu-role actionAbout about
wait 400
expect messageWindow About EasyRPG Editor
expect messageText A game editor for EasyRPG
close
wait 200

menu-role actionSettings preferences
wait 400
expect SettingsDialog Settings
close
wait 200

menu-role actionPlayTestOptions application
wait 400
expect RunGameDialog Test Play Options
close
wait 200

# Quit asks about the unsaved maps first: Cancel, the Editor stays
open-map 1
action actionLayerLower
action actionDrawPen
palette 7
map-click 1 1
menu-role actionQuit quit
wait 400
click Cancel
wait 200
expect-enabled actionMapSave on
expect treeMap RPG Maker 2003 Test suite
quit
