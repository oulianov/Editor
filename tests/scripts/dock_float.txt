# A dock floated (in a window of its own) is floating again when its
# project opens again (its workspace), whatever it was meanwhile; docked,
# docked again. On a copy of the project: nothing is saved.
# Run on TestGame-2003 (https://github.com/EasyRPG/TestGame).
open-project-copy
wait 300
expect-dock dockPalette docked
expect-dock dockMapTree docked
float-dock dockPalette on
wait 300
expect-dock dockPalette floating
expect-dock dockMapTree docked
action actionProjectClose
wait 300
float-dock dockPalette off
wait 300
expect-dock dockPalette docked
open-project
wait 400
expect-dock dockPalette floating
expect-dock dockMapTree docked
expect-checked actionPaletteToggle on

# both floating, then both back
float-dock dockMapTree on
wait 300
action actionProjectClose
wait 300
float-dock dockMapTree off
wait 300
open-project
wait 400
expect-dock dockPalette floating
expect-dock dockMapTree floating
float-dock dockPalette off
float-dock dockMapTree off
wait 300
action actionProjectClose
wait 300
float-dock dockPalette on
wait 300
open-project
wait 400
expect-dock dockPalette docked
expect-dock dockMapTree docked
quit
