# Tools > Jukebox (QML): the music and the sound effects of the game (and
# of the RTP, when the settings have one), filtered, the database's lines,
# RPG Maker's settings, a file chosen (not played: the tests make no
# sound), and the window kept as it was when opened again.
action actionJukebox
wait 500
expect JukeboxDialog Jukebox
expect tabsJukebox Music | Sound Effects
expect buttonBox Close
check-layout
expect listMusic Battle 1
# the System tab of the database
expect listMusic Title Screen:
expect-checked checkLoop on
expect-enabled groupFadeIn on
expect-enabled buttonStop off
expect labelTime 0:00

# the filter: both lists
set-control editFilter Castle
expect listMusic Castle 1
expect-not listMusic Battle 1
set-control editFilter ""
expect listMusic Battle 1

# the sound effects: no fade-in, played once
tab tabsJukebox 1
expect-checked checkLoop off
expect-enabled groupFadeIn off
expect listSound Cursor:
tab tabsJukebox 0
expect-checked checkLoop on

# RPG Maker's settings
set-control sliderVolume 50
expect labelVolume 50%
set-control sliderTempo 150
expect labelTempo 150%
# RPG Maker's default (100%) back: the reset button then greyed
click resetTempo
expect labelTempo 100%
expect-enabled resetTempo off
set-control sliderBalance 25
expect labelBalance 75:25
set-control sliderFadeIn 3
expect labelFadeIn 3 s

# a file chosen: it can be played (it is not: no sound in the tests)
click Cat
wait 200
expect listMusic > Cat
expect-enabled buttonPlay on
expect-enabled buttonStop off
expect-enabled buttonPause off
close
wait 300

# opened again: as it was left
action actionJukebox
wait 500
expect listMusic > Cat
expect labelVolume 50%
expect labelBalance 75:25
close
quit
