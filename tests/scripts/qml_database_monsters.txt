# The Monsters page of the QML database window (on a copy of the project):
# the statistics and spoils, a resistance, the behavior edited in its
# window (an action added on the empty line, one cut and pasted); OK writes
# them, Cancel writes nothing.
open-project-copy
action actionDatabase
wait 300
tab tabOld_Pages 4
wait 300
expect-tab tabOld_Pages Monsters
expect list > 0001: Ariman
expect lineName Ariman
expect spinMaxHp 5625
expect spinAttack 480
expect comboDropItem (None)
expect-enabled spinCritChance off
expect treeActions Attack | No Precondition | 50
expect treeActions Blizzard Breath | No Precondition | 50
expect listAttributeRanks rank3 | Darkness
check-layout
resize 1280 900
check-layout

set-control lineName Lord
set-control spinMaxHp 7000
set-control spinExp 15
set-control comboDropItem 1
set-control checkCritical on
expect-enabled spinCritChance on
set-control spinCritChance 5
set-control checkMiss on
focus listAttributeRanks
key Home
key Space
expect listAttributeRanks rank3 | Sword
# a new action on the empty line: a skill, from turn 3 every 2 turns
focus treeActions
key End
key Return
expect EnemyActionDialog Monster Behavior
expect-checked radioBasic on
expect-enabled comboSkill off
set-control comboCondition 2
expect spinParam1 1
set-control spinParam1 2
set-control spinParam2 3
set-control spinRating 70
set-control radioSkill on
expect-enabled comboSkill on
set-control comboSkill 1
set-control checkSwitchOn on
set-control comboSwitchOn 3
click OK
expect treeActions Blind | Turns Elapsed [2x+3] | 70
# Cancel of a new action adds nothing
key End
key Return
click Cancel
expect-not treeActions Attack | No Precondition | 70
# the first action cut, pasted before the last line
key Home
key Ctrl+X
expect-not treeActions Attack | No Precondition | 50
key End
key Up
key Ctrl+V
click OK
wait 300
expect-database-file enemies/0/name Lord
expect-database-file enemies/0/max_hp 7000
expect-database-file enemies/0/exp 15
expect-database-file enemies/0/drop_id 1
expect-database-file enemies/0/critical_hit true
expect-database-file enemies/0/critical_hit_chance 5
expect-database-file enemies/0/miss true
expect-database-file enemies/0/attribute_ranks/0 3
expect-database-file enemies/0/actions/3/kind 0
expect-database-file enemies/0/actions/3/ID 4
expect-database-file enemies/0/actions/4/kind 1
expect-database-file enemies/0/actions/4/skill_id 2
expect-database-file enemies/0/actions/4/condition_type 2
expect-database-file enemies/0/actions/4/condition_param1 2
expect-database-file enemies/0/actions/4/condition_param2 3
expect-database-file enemies/0/actions/4/rating 70
expect-database-file enemies/0/actions/4/switch_on true
expect-database-file enemies/0/actions/4/switch_on_id 4
expect-database-file enemies/0/actions/4/ID 5

# Cancel: nothing written
action actionDatabase
wait 300
expect-tab tabOld_Pages Monsters
expect lineName Lord
set-control lineName Nobody
focus treeActions
key Home
key Del
close
wait 300
expect-database-file enemies/0/name Lord
expect-database-file enemies/0/actions/0/kind 1
quit
